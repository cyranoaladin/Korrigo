        'NAME': 'django.contrib.auth.password_validation.UserAttributeSimilarityValidator',

    {
        'NAME': 'django.contrib.auth.password_validation.MinimumLengthValidator',

        'NAME': 'django.contrib.auth.password_validation.CommonPasswordValidator',

### 2. Authentification Étudiants

**Méthode** : Email + Date de naissance (session-based, pas de compte Django)

**Sécurité** :

```python
# students/views.py

from django_ratelimit.decorators import ratelimit

@ratelimit(key='ip', rate='5/m', method='POST')
@api_view(['POST'])
def student_login(request):
    """
    Login étudiant avec rate limiting.

    Rate limit: 5 tentatives par minute par IP
    """
    email = request.data.get('email')
    dob = request.data.get('date_of_birth')

    # Validation
    if not email or not dob:
        return Response({'error': 'Missing credentials'}, status=400)

    # Recherche étudiant (case-insensitive email)
    try:
        student = Student.objects.get(
            email__iexact=email,
            date_of_birth=parse_date(dob)
        )
    except Student.DoesNotExist:
        # Log failed attempt
        logger.warning(f"Failed student login: {email} from {request.META.get('REMOTE_ADDR')}")

        # Generic error message (éviter énumération)
        return Response({'error': 'Invalid credentials'}, status=401)

    # Créer session
    request.session['student_id'] = student.id
    request.session['is_student'] = True
    request.session.set_expiry(3600 * 2)  # 2 heures

    return Response({'success': True})
```

**Risques atténués** :

| Risque | Mitigation |
|--------|------------|
| **Brute-force** | Rate limiting (5 tentatives/minute) |
| **Énumération emails** | Message erreur générique |
| **Session hijacking** | HTTPS + HttpOnly + SameSite cookies |
| **Date naissance publique** | Session courte (2h), lecture seule |

## 🛡️ Autorisations (Permissions)

### Modèle RBAC (Role-Based Access Control)

**Rôles** :

| Rôle | Permissions | Cas d'usage |
|------|-------------|-------------|
| **ADMIN** | Tous droits | Administrateur système |
| **TEACHER** | CRUD examens, correction, publication | Enseignant |
| **STUDENT** | Lecture copies publiées (soi-même uniquement) | Élève |

### Permissions Django REST Framework

```python
# core/permissions.py

from rest_framework.permissions import BasePermission

class IsAdmin(BasePermission):
    """Uniquement admins."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated and request.user.role == 'ADMIN'


class IsTeacher(BasePermission):
    """Admins + Teachers."""

    def has_permission(self, request, view):
        return request.user and request.user.is_authenticated and request.user.role in ['ADMIN', 'TEACHER']


class IsExamOwner(BasePermission):
    """Uniquement créateur examen."""

    def has_object_permission(self, request, view, obj):
        if not request.user or not request.user.is_authenticated:
            return False

        # Admin → accès total
        if request.user.role == 'ADMIN':
            return True

        # Teacher → uniquement ses examens
        return obj.created_by == request.user


class IsCopyOwner(BasePermission):
    """Uniquement étudiant propriétaire copie."""

    def has_object_permission(self, request, view, obj):
        student_id = request.session.get('student_id')

        if not student_id:
            return False

        # Copie appartient à l'étudiant
        return obj.student_id == student_id
```

class ExamViewSet(viewsets.ModelViewSet):

## 🔒 Protection CSRF

### Configuration

```python
# backend/settings.py

# CSRF Settings
CSRF_COOKIE_NAME = 'korrigo_csrftoken'
CSRF_COOKIE_AGE = 31449600  # 1 an
CSRF_COOKIE_SECURE = True  # HTTPS uniquement
CSRF_COOKIE_HTTPONLY = False  # JavaScript doit lire le cookie
CSRF_COOKIE_SAMESITE = 'Strict'

# Trusted Origins
CSRF_TRUSTED_ORIGINS = [
    'https://korrigo.yourdomain.com',
    'https://www.korrigo.yourdomain.com'
]
```

### Frontend (Axios)

```typescript
// frontend/src/services/api.ts

import axios from 'axios'
import Cookies from 'js-cookie'

const api = axios.create({
  baseURL: '/api',
  withCredentials: true,  // Envoyer cookies
  headers: {
    'Content-Type': 'application/json'
  }
})

// Interceptor: Ajouter CSRF token à chaque requête
api.interceptors.request.use(config => {
  const csrfToken = Cookies.get('korrigo_csrftoken')

  if (csrfToken && ['post', 'put', 'patch', 'delete'].includes(config.method?.toLowerCase() || '')) {
    config.headers['X-CSRFToken'] = csrfToken
  }

  return config
})

export default api
```

## 🚫 Protection Injection

### 1. SQL Injection

**Django ORM** : Protection automatique via paramétrage requêtes

**BON** (paramétré) :

```python
# ✅ Sécurisé
Student.objects.filter(email=user_input)

# ✅ Sécurisé
Student.objects.raw('SELECT * FROM students WHERE email = %s', [user_input])
```

**MAUVAIS** (concaténation) :

```python
# ❌ Vulnérable SQL injection
Student.objects.raw(f'SELECT * FROM students WHERE email = "{user_input}"')

# ❌ Vulnérable
cursor.execute(f"SELECT * FROM students WHERE email = '{user_input}'")
```

### 2. XSS (Cross-Site Scripting)

**Frontend Vue.js** : Échappement automatique

```vue
<!-- ✅ Échappé automatiquement -->
<p>{{ userInput }}</p>

<!-- ❌ Dangereux: HTML brut -->
<p v-html="userInput"></p>
```

**Sanitize HTML si nécessaire** :

```typescript
import DOMPurify from 'dompurify'

const cleanHTML = DOMPurify.sanitize(userInput)
```

**Backend Django** : Échappement automatique dans templates

```django
<!-- ✅ Échappé -->
<p>{{ user_comment }}</p>

<!-- ❌ Dangereux -->
<p>{{ user_comment|safe }}</p>
```

### 3. Command Injection

**Éviter `os.system()`, `subprocess.shell=True`**

**BON** :

```python
# ✅ Sécurisé (liste arguments)
import subprocess

result = subprocess.run(
    ['pdfinfo', pdf_path],
    capture_output=True,
    text=True,
    check=True
)
```

**MAUVAIS** :

```python
# ❌ Vulnérable command injection
os.system(f'pdfinfo {pdf_path}')

# ❌ Vulnérable
subprocess.run(f'pdfinfo {pdf_path}', shell=True)
```

## 📊 Rate Limiting

### Configuration django-ratelimit

**Installation** :

```bash
pip install django-ratelimit
```

**Zones** :

| Endpoint | Rate | Burst | Clé |
|----------|------|-------|-----|
| Login étudiant | 5/minute | - | IP |
| API générale | 100/minute | 20 | User |
| Upload PDF | 10/heure | 2 | User |

**Implémentation** :

```python
# views.py

@ratelimit(key='ip', rate='5/m', method='POST')
@api_view(['POST'])
def student_login(request):
    """Login avec rate limiting."""
    # Si rate limit dépassé, django-ratelimit retourne 429
    # ...
```

**Custom decorator** :

```python
# core/decorators.py

from django_ratelimit.decorators import ratelimit
from functools import wraps

def api_rate_limit(rate='100/m'):
    """Rate limiting API avec clé user ou IP."""

    def decorator(func):
        @wraps(func)
        @ratelimit(key='user_or_ip', rate=rate, method=['GET', 'POST', 'PUT', 'PATCH', 'DELETE'])
        def wrapper(request, *args, **kwargs):
            return func(request, *args, **kwargs)
        return wrapper
    return decorator

# Usage
@api_rate_limit(rate='30/m')
@api_view(['GET'])
def some_api_view(request):
    # ...
```

## 📁 Sécurité Uploads

### Validation Fichiers

```python
# exams/validators.py

from django.core.exceptions import ValidationError
import magic
import PyPDF2

def validate_pdf_file(file):
    """Valider fichier PDF uploadé."""

    # 1. Vérifier extension
    if not file.name.lower().endswith('.pdf'):
        raise ValidationError('Invalid file type. Must be PDF.')

    # 2. Vérifier taille (100 MB max)
    max_size = 100 * 1024 * 1024
    if file.size > max_size:
        raise ValidationError(f'File too large: {file.size} bytes (max: {max_size})')

    # 3. Vérifier MIME type (magic bytes)
    file.seek(0)
    mime = magic.from_buffer(file.read(1024), mime=True)
    if mime != 'application/pdf':
        raise ValidationError(f'Invalid file type: {mime}. Expected application/pdf')

    # 4. Vérifier intégrité PDF
    file.seek(0)
    try:
        pdf_reader = PyPDF2.PdfReader(file)
        page_count = len(pdf_reader.pages)

        if page_count == 0:
            raise ValidationError('PDF has no pages')

    except Exception as e:
        raise ValidationError(f'Corrupted PDF: {str(e)}')

    # Reset file pointer
    file.seek(0)


def validate_csv_file(file):
    """Valider fichier CSV uploadé."""

    # Extension
    if not file.name.lower().endswith('.csv'):
        raise ValidationError('Invalid file type. Must be CSV.')

    # Taille (10 MB max)
    max_size = 10 * 1024 * 1024
    if file.size > max_size:
        raise ValidationError(f'File too large: {file.size} bytes (max: {max_size})')

    # MIME type
    file.seek(0)
    mime = magic.from_buffer(file.read(1024), mime=True)
    if mime not in ['text/plain', 'text/csv', 'application/csv']:
        raise ValidationError(f'Invalid file type: {mime}')

    file.seek(0)
```

## 🧪 Tests Sécurité

### Test Permissions

```python
# backend/core/tests/test_permissions.py

from django.test import TestCase
from rest_framework.test import APIClient
from django.contrib.auth import get_user_model

User = get_user_model()

class PermissionsTestCase(TestCase):
    def setUp(self):
        self.client = APIClient()

        # Créer users
        self.admin = User.objects.create_user(username='admin', role='ADMIN', password='pass')
        self.teacher = User.objects.create_user(username='teacher', role='TEACHER', password='pass')
        self.other_teacher = User.objects.create_user(username='other', role='TEACHER', password='pass')

    def test_teacher_cannot_modify_other_exam(self):
        """Test qu'un teacher ne peut pas modifier examen d'un autre."""
        # Créer examen par teacher
        self.client.force_authenticate(user=self.teacher)
        response = self.client.post('/api/exams/', {
            'title': 'Test Exam',
            'subject': 'Maths'
        })
        exam_id = response.data['id']

        # Tenter modification par other_teacher
        self.client.force_authenticate(user=self.other_teacher)
        response = self.client.patch(f'/api/exams/{exam_id}/', {
            'title': 'Hacked Title'
        })

        self.assertEqual(response.status_code, 403)

    def test_student_cannot_access_unpublished_copy(self):
        """Test qu'un étudiant ne peut pas voir copie non publiée."""
        # Créer copie GRADED (non publiée)
        copy = Copy.objects.create(
            exam=self.exam,
            student=self.student,
            status='GRADED'
        )

        # Login étudiant
        session = self.client.session
        session['student_id'] = self.student.id
        session['is_student'] = True
        session.save()

        # Tenter accès
        response = self.client.get(f'/api/students/copies/{copy.id}/')

        self.assertEqual(response.status_code, 403)
```
